📊 ESTRUCTURA DE DATOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

{
  "id": 1,
  "producto_padre_id": 5,
  "producto_hijo_id": 10,
  "cantidad": 2.50,
  "estado": 1
}

Validaciones requeridas:
  ✓ producto_padre_id exists in productos.id
  ✓ producto_hijo_id exists in productos.id
  ✓ producto_padre_id ≠ producto_hijo_id
  ✓ cantidad > 0
  ✓ estado in [0, 1]

💾 TABLA MYSQL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CREATE TABLE `composicion_productos` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `producto_padre_id` bigint unsigned NOT NULL,
  `producto_hijo_id` bigint unsigned NOT NULL,
  `cantidad` decimal(12,2) NOT NULL,
  `estado` int NOT NULL DEFAULT 1,
  `created_at` timestamp,
  `updated_at` timestamp,
  FOREIGN KEY (`producto_padre_id`) REFERENCES `productos`(`id`) 
    ON DELETE CASCADE,
  FOREIGN KEY (`producto_hijo_id`) REFERENCES `productos`(`id`) 
    ON DELETE CASCADE
);
